
--// 1 PLAYER HUB
--// Compact animated player-list GUI
--// Place inside StarterPlayer > StarterPlayerScripts

local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

local old = playerGui:FindFirstChild("OnePlayerHub")
if old then old:Destroy() end

local gui = Instance.new("ScreenGui")
gui.Name = "OnePlayerHub"
gui.ResetOnSpawn = false
gui.Parent = playerGui

local function new(class, props, parent)
    local obj = Instance.new(class)
    for k, v in pairs(props) do obj[k] = v end
    obj.Parent = parent
    return obj
end

local main = new("Frame", {
    Size = UDim2.fromOffset(245, 275),
    Position = UDim2.new(0.5, -122, 0.5, -137),
    BackgroundColor3 = Color3.fromRGB(10, 12, 22),
    BorderSizePixel = 0,
    ClipsDescendants = true
}, gui)

new("UICorner", {CornerRadius = UDim.new(0, 10)}, main)

local stroke = new("UIStroke", {
    Color = Color3.fromRGB(0, 200, 255),
    Thickness = 1.5
}, main)

-- Animated background
local gradient = new("UIGradient", {
    Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(5, 10, 35)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(20, 5, 45)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 45, 60))
    }),
    Rotation = 0
}, main)

task.spawn(function()
    while main.Parent do
        TweenService:Create(gradient, TweenInfo.new(3), {
            Rotation = 360
        }):Play()
        TweenService:Create(stroke, TweenInfo.new(1.5), {
            Color = Color3.fromRGB(255, 35, 120)
        }):Play()
        task.wait(1.5)
        if not main.Parent then break end
        TweenService:Create(stroke, TweenInfo.new(1.5), {
            Color = Color3.fromRGB(0, 220, 255)
        }):Play()
        task.wait(1.5)
    end
end)

-- Header
local header = new("Frame", {
    Size = UDim2.new(1, 0, 0, 35),
    BackgroundColor3 = Color3.fromRGB(15, 18, 35),
    BackgroundTransparency = 0.1,
    BorderSizePixel = 0
}, main)

local title = new("TextLabel", {
    Size = UDim2.new(1, -90, 1, 0),
    Position = UDim2.fromOffset(10, 0),
    BackgroundTransparency = 1,
    Text = "✦ 1 PLAYER HUB",
    TextColor3 = Color3.fromRGB(0, 225, 255),
    TextSize = 14,
    Font = Enum.Font.GothamBold,
    TextXAlignment = Enum.TextXAlignment.Left
}, header)

local function makeButton(text, pos, size, parent)
    local b = new("TextButton", {
        Size = size,
        Position = pos,
        Text = text,
        TextSize = 14,
        Font = Enum.Font.GothamBold,
        TextColor3 = Color3.new(1, 1, 1),
        BackgroundColor3 = Color3.fromRGB(35, 40, 65),
        BorderSizePixel = 0,
        AutoButtonColor = false
    }, parent)
    new("UICorner", {CornerRadius = UDim.new(0, 5)}, b)

    b.MouseEnter:Connect(function()
        TweenService:Create(b, TweenInfo.new(0.18), {
            BackgroundColor3 = Color3.fromRGB(0, 130, 180)
        }):Play()
    end)
    b.MouseLeave:Connect(function()
        TweenService:Create(b, TweenInfo.new(0.18), {
            BackgroundColor3 = Color3.fromRGB(35, 40, 65)
        }):Play()
    end)
    return b
end

local minimize = makeButton("—", UDim2.new(1, -59, 0, 6),
    UDim2.fromOffset(23, 23), header)
local close = makeButton("×", UDim2.new(1, -30, 0, 6),
    UDim2.fromOffset(23, 23), header)

-- Status
local status = new("TextLabel", {
    Size = UDim2.new(1, -16, 0, 20),
    Position = UDim2.fromOffset(8, 39),
    BackgroundTransparency = 1,
    Text = "SERVER PLAYERS",
    TextColor3 = Color3.fromRGB(170, 185, 220),
    TextSize = 10,
    Font = Enum.Font.GothamBold,
    TextXAlignment = Enum.TextXAlignment.Left
}, main)

-- Scrollable player list
local list = new("ScrollingFrame", {
    Size = UDim2.new(1, -16, 1, -103),
    Position = UDim2.fromOffset(8, 62),
    BackgroundColor3 = Color3.fromRGB(8, 10, 20),
    BackgroundTransparency = 0.15,
    BorderSizePixel = 0,
    ScrollBarThickness = 3,
    ScrollBarImageColor3 = Color3.fromRGB(0, 220, 255),
    CanvasSize = UDim2.new(0, 0, 0, 0),
    AutomaticCanvasSize = Enum.AutomaticSize.Y

    PaddingTop = UDim.new(0, 5),

    PaddingRight = UDim.new(0, 5),

}, list)

local refresh = makeButton("↻  REFRESH PLAYERS",
    UDim2.new(0, 8, 1, -34),
    UDim2.new(1, -16, 0, 26), main)

local function updatePlayers()
    for _, child in ipairs(list:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end

    local all = Players:GetPlayers()
    table.sort(all, function(a, b)
        return a.Name:lower() < b.Name:lower()
    end)

    status.Text = "SERVER PLAYERS  •  " .. #all

    for i, plr in ipairs(all) do
        local age = plr.AccountAge
        local isNew = age <= 7

        local row = new("Frame", {
            Size = UDim2.new(1, -2, 0, 38),
            BackgroundColor3 = Color3.fromRGB(20, 25, 42),
            BorderSizePixel = 0,
            LayoutOrder = i
        }, list)

        new("UICorner", {CornerRadius = UDim.new(0, 5)}, row)

        local avatar = new("ImageLabel", {
            Size = UDim2.fromOffset(28, 28),
            Position = UDim2.fromOffset(5, 5),
            BackgroundTransparency = 1,
            Image = ""
        }, row)

        task.spawn(function()
            local ok, image = pcall(function()
                return Players:GetUserThumbnailAsync(
                    plr.UserId,
                    Enum.ThumbnailType.HeadShot,
                    Enum.ThumbnailSize.Size100x100
                )
            end)
            if ok and avatar.Parent then avatar.Image = image end
        end)

        new("TextLabel", {
            Size = UDim2.new(1, -100, 1, 0),
            Position = UDim2.fromOffset(38, 0),
            BackgroundTransparency = 1,
            Text = plr.DisplayName,
            TextColor3 = Color3.fromRGB(240, 245, 255),
            TextSize = 11,
            Font = Enum.Font.GothamBold,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextTruncate = Enum.TextTruncate.AtEnd
        }, row)

        new("TextLabel", {
            Size = UDim2.fromOffset(57, 14),
            Position = UDim2.new(1, -62, 0, 3),
            BackgroundTransparency = 1,
            Text = isNew and "NEW ACCOUNT" or (age .. "d old"),
            TextColor3 = isNew
                and Color3.fromRGB(0, 255, 160)
                or Color3.fromRGB(150, 170, 200),
            TextSize = 8,
            Font = Enum.Font.GothamBold
        }, row)

        new("TextLabel", {
            Size = UDim2.new(1, -100, 0, 13),
            Position = UDim2.fromOffset(38, 20),
            BackgroundTransparency = 1,
            Text = "@" .. plr.Name,
            TextColor3 = Color3.fromRGB(130, 150, 180),
            TextSize = 9,
            Font = Enum.Font.Gotham
        }, row)

        row.BackgroundTransparency = 1
        TweenService:Create(row, TweenInfo.new(
            0.25 + i * 0.025,
            Enum.EasingStyle.Quad,
            Enum.EasingDirection.Out
        ), {BackgroundTransparency = 0}):Play()
    end
end

refresh.MouseButton1Click:Connect(function()
    refresh.Text = "LOADING..."
    updatePlayers()
    task.wait(0.3)
    if refresh.Parent then refresh.Text = "↻  REFRESH PLAYERS" end
end)

Players.PlayerAdded:Connect(updatePlayers)
Players.PlayerRemoving:Connect(updatePlayers)

-- Minimize / restore
local minimized = false
minimize.MouseButton1Click:Connect(function()
    minimized = not minimized
    if minimized then
        TweenService:Create(main, TweenInfo.new(0.25), {
            Size = UDim2.fromOffset(245, 35)
        }):Play()
        minimize.Text = "+"
    else
        TweenService:Create(main, TweenInfo.new(0.25), {
            Size = UDim2.fromOffset(245, 275)
        }):Play()
        minimize.Text = "—"
    end
end)

-- Close button
close.MouseButton1Click:Connect(function()
    gui:Destroy()
end)

-- Drag with mouse or touch
local dragging = false
local dragStart, startPos
local dragInput

header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

header.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (
        input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch
    ) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

-- Entrance animation
main.Size = UDim2.fromOffset(0, 0)
TweenService:Create(main, TweenInfo.new(
    0.55,
    Enum.EasingStyle.Back,
    Enum.EasingDirection.Out
), {Size = UDim2.fromOffset(245, 275)}):Play()

updatePlayers()
